cmake_minimum_required(VERSION 3.22)
project(nexus_llm LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS nexus_compile=2 nexus_link=1)
set(CMAKE_JOB_POOL_COMPILE nexus_compile)
set(CMAKE_JOB_POOL_LINK nexus_link)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
if(ANDROID)
  set(GGML_AVX OFF CACHE BOOL "" FORCE)
  set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
  set(GGML_FMA OFF CACHE BOOL "" FORCE)
  set(GGML_F16C OFF CACHE BOOL "" FORCE)
  set(GGML_BMI2 OFF CACHE BOOL "" FORCE)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED NEXUS_LLAMA_SOURCE)
  set(NEXUS_LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../build/llama-source")
endif()
if(NOT ANDROID OR ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "x86_64")
  add_subdirectory("${NEXUS_LLAMA_SOURCE}" "${CMAKE_BINARY_DIR}/llama")
  add_library(nexus_llm SHARED nexus_llm.cpp)
  target_link_libraries(nexus_llm PRIVATE llama)
else()
  add_library(nexus_llm SHARED unavailable.cpp)
endif()
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(nexus_llm PRIVATE ${JNI_INCLUDE_DIRS})
endif()
if(ANDROID)
  target_link_options(nexus_llm PRIVATE "-Wl,-z,max-page-size=16384")
endif()
